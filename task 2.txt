TASK 2: Medium – Matrix Operations

#include <stdio.h>

#define MAX 10

void inputMatrix(int mat[MAX][MAX], int row, int col) {
    for (int i = 0; i < row; i++) {
        for (int j = 0; j < col; j++) {
            printf("Element [%d][%d]: ", i + 1, j + 1);
            scanf("%d", &mat[i][j]);
        }
    }
}

void displayMatrix(int mat[MAX][MAX], int row, int col) {
    for (int i = 0; i < row; i++) {
        for (int j = 0; j < col; j++) {
            printf("%d\t", mat[i][j]);
        }
        printf("\n");
    }
}

void addMatrices(int a[MAX][MAX], int b[MAX][MAX], int row, int col) {
    int sum[MAX][MAX];
    for (int i = 0; i < row; i++) {
        for (int j = 0; j < col; j++) {
            sum[i][j] = a[i][j] + b[i][j];
        }
    }
    printf("\nResult of Matrix Addition:\n");
    displayMatrix(sum, row, col);
}

void multiplyMatrices(int a[MAX][MAX], int b[MAX][MAX], int r1, int c1, int c2) {
    int prod[MAX][MAX] = {0};
    for (int i = 0; i < r1; i++) {
        for (int j = 0; j < c2; j++) {
            for (int k = 0; k < c1; k++) {
                prod[i][j] += a[i][k] * b[k][j];
            }
        }
    }
    printf("\nResult of Matrix Multiplication:\n");
    displayMatrix(prod, r1, c2);
}

void transposeMatrix(int mat[MAX][MAX], int row, int col) {
    int trans[MAX][MAX];
    for (int i = 0; i < row; i++) {
        for (int j = 0; j < col; j++) {
            trans[j][i] = mat[i][j];
        }
    }
    printf("\nTranspose of the Matrix:\n");
    displayMatrix(trans, col, row);
}

int main() {
    int a[MAX][MAX], b[MAX][MAX];
    int r1, c1, r2, c2, choice;

    printf("1. Matrix Addition\n2. Matrix Multiplication\n3. Matrix Transpose\nEnter choice: ");
    scanf("%d", &choice);

    if (choice == 1) {
        printf("Enter rows and columns for matrices: ");
        scanf("%d %d", &r1, &c1);
        printf("Enter elements of Matrix A:\n");
        inputMatrix(a, r1, c1);
        printf("Enter elements of Matrix B:\n");
        inputMatrix(b, r1, c1);
        addMatrices(a, b, r1, c1);
    } 
    else if (choice == 2) {
        printf("Enter rows and columns for Matrix A: ");
        scanf("%d %d", &r1, &c1);
        printf("Enter rows and columns for Matrix B: ");
        scanf("%d %d", &r2, &c2);

        if (c1 != r2) {
            printf("Error: Multiplication not possible (Columns of A != Rows of B).\n");
            return 1;
        }
        printf("Enter elements of Matrix A:\n");
        inputMatrix(a, r1, c1);
        printf("Enter elements of Matrix B:\n");
        inputMatrix(b, r2, c2);
        multiplyMatrices(a, b, r1, c1, c2);
    } 
    else if (choice == 3) {
        printf("Enter rows and columns of the matrix: ");
        scanf("%d %d", &r1, &c1);
        printf("Enter elements of the Matrix:\n");
        inputMatrix(a, r1, c1);
        transposeMatrix(a, r1, c1);
    } 
    else {
        printf("Invalid choice.\n");
    }

    return 0;
}
